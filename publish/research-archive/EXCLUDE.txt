# research-archive NEVER-INCLUDED paths (Stage P.1, D4). git-filter-repo --paths-from-file syntax, used with
# --invert-paths on each source's ORIGINAL paths, before --to-subdirectory-filter. Tested 2026-11-07 against every
# path in the full history of every source (reports/P.1_STATE.md).
# Two D4 rules test content, not paths. build_archive.sh applies them in a pre-pass over every (path, blob) in a
# source's full history, adds each hit path to that source's exclusion list, and logs it in the build report:
#   R1 any JSON that parses as an array of 64 integers 0-255 (a wallet keypair file), whatever its name or place
#   R2 any blob over 1 MB (1048576 bytes) that is not code or markdown written in the repo
# Tiny fixtures that tests need stay only if proven synthetic. No source has one: MLCryptoEngine's
# tests/fixtures/README.md says its messages were "recorded live" with "no synthetic data". Tests that need
# them are documented, not fixed (D9).

# Data and model files
regex:(?i)(^|.*/)[^/]*\.(db|db-[^/]*|sqlite[^/]*|parquet|feather|pkl|pickle|npy|npz|h5|hdf5|arrow|ipc)$
regex:(?i)(^|.*/)[^/]*\.(pt|pth|ckpt|safetensors|onnx|joblib|gguf|ggml|tflite|keras|weights|bin|pb|model)$
# Compressed archives can hold data of any kind
regex:(?i)(^|.*/)[^/]*\.(zip|gz|tgz|tar|bz2|xz|7z|rar|zst)$
# Data folders anywhere: every file under data/, raw/, cache/, logs/ or log/, except code and markdown.
# MLCryptoEngine data/ and NPDExperiment src/npd/data/ are Python packages, so their .py files stay.
regex:(?i)(^|.*/)(data|raw|cache|\.cache|logs?)/(.*/)?[^/]+(?<!\.py)(?<!\.rs)(?<!\.ts)(?<!\.js)(?<!\.sh)(?<!\.md)$
# Recorded vendor data used as test fixtures (code and markdown fixtures stay)
regex:(?i)(^|.*/)fixtures?/(.*/)?[^/]+\.(json|jsonl|ndjson|csv|tsv|html?|txt|xml|bin)$
# program-audit work/: agent scratch with copies of other repos and bulk on-chain data (1,359 paths)
work/

# Secrets: env files (templates included), key and keystore files, and DATA files named like a key, wallet,
# secret or token. Code files with such names (for example ORBExperiment src/orb/credentials.py) stay and are
# checked by the D6 content scans instead.
regex:(?i)(^|.*/)\.env[^/]*$
regex:(?i)(^|.*/)[^/]*\.(pem|key|p12|pfx|jks|keystore|asc|gpg|kdbx)$
regex:(?i)(^|.*/)[^/]*(key|wallet|secret|token|credential|mnemonic|seed|keystore|keypair|id_rsa|id_ed25519)[^/]*\.(json|txt|b58|dat|yaml|yml|toml|ini|cfg|conf|csv)$
regex:(?i)(^|.*/)[^/.]*(key|wallet|secret|token|credential|mnemonic|seed|keystore|keypair|id_rsa|id_ed25519)[^/.]*$

# Saved third-party pages. Two HTML files were written in their repos and stay (checked 2026-11-07):
# MLCryptoEngine desktop/index.html and NPDExperiment reports/session_record.html.
regex:(?i)(^|.*/)[^/]*\.(mhtml|mht|pdf|webarchive)$
regex:(?i)^(?!desktop/index\.html$|reports/session_record\.html$)(.*/)?[^/]*\.html?$

# Vendor market data by name: price, quote, trade, option, bar, tick, candle, kline, chain or order book tables
regex:(?i)(^|.*/)[^/]*(price|quote|trade|option|ohlc|bars|tick|candle|kline|chain|orderbook|order_book)[^/]*\.(csv|tsv|jsonl|ndjson|txt)$
